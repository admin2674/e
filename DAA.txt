def merge_sort(arr):
    if len(arr) <= 1:
        return arr

    mid = len(arr) // 2
    lefthalf = arr[:mid]
    righthalf = arr[mid:]

    print("Dividing: Left=", lefthalf, "Right=", righthalf)

    sortedleft = merge_sort(lefthalf)
    sortedright = merge_sort(righthalf)

    return merge(sortedleft, sortedright)

def merge(left, right):
    result = []
    i = j = 0

    while i < len(left) and j < len(right):
        if left[i] < right[j]:
            result.append(left[i])
            i += 1
        else:
            result.append(right[j])
            j += 1
        
    result.extend(left[i:])
    result.extend(right[j:])

    print("Merged: ", result)
    return result

array = [11, 8, 4, 16, 2, 23]
print("Original array : ", array)
sortedArray = merge_sort(array)
print("Sorted array: ", sortedArray)

<!------------------------------ Quick Sort ----------------------------------------------->

def quick_sort(arr, low, high):
    if low < high:
        pivot = partition(arr, low, high)
        quick_sort(arr, low, pivot - 1)
        quick_sort(arr, pivot + 1, high)

def partition(arr, low, high):
    p = arr[low]
    print("Pivot chosen: ", p, "in range [", low, "..", high, "]")
    i = low + 1
    j = high

    while True:
        while i <= j and arr[i] <= p:
            i += 1
        while i <= j and arr[j] >= p:
            j -= 1
        if i <= j:
            arr[i], arr[j] = arr[j], arr[i]
        else:
            break

    arr[low], arr[j] = arr[j], arr[low]
    print("State after partitioning around ", p, ": ", arr)
    return j

arr = [7, 1, 2, 5, 6, 8]
quick_sort(arr, 0, len(arr) - 1)
print("Sorted array : ", arr)

def quick_sort_last_pivot(arr, low, high):
    if low < high:
        pivot = partition(arr, low, high)
        quick_sort_last_pivot(arr, low, pivot - 1)
        quick_sort_last_pivot(arr, pivot + 1, high)

def partition_last_pivot(arr, low, high):
    p = arr[high]
    print("Pivot chosen: ", p, "in range [", low, "..", high, "]")
    i = low - 1
    for j in range(low, high):
        if arr[j] <= p:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]

    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    print("State after partitioning around ", p, ": ", arr)
    return i + 1

<!----------------------------- Activity Selection Problem ------------------->

def activity_selection(start, finish):
    activities = list(zip(start, finish))
    activities.sort(key=lambda x: x[1])
    
    print("Sorted Activities (Start, Finish):", activities)
    print("-" * 45)
    
    selected = []
    last_end = 0
    step = 1 

    for s, f in activities:
        if s >= last_end:
            selected.append((s, f))
            print("Step ", step, ": Selected (",s,",",f,") | New Last End Time = ", f)
            last_end = f
        else:
            print("Step ", step, ": Skipped  (",s,",",f,") | Conflicts with Last End Time (",last_end,")")
        step += 1 
    return selected

start = [1, 3, 0, 5, 8, 5]
finish = [2, 4, 6, 7, 9, 9]

result = activity_selection(start, finish)
print("-" * 45)
print("Final Selected Activities : ", result)


<!----------------------------- fractinal knapsack problem ------------------->

def fractional_knapsack(capacity, weights, values):
    n = len(values)
    items = []
    for i in range(n):
        ratio = values[i] / weights[i]
        items.append((ratio, values[i], weights[i]))
    items.sort(reverse=True)
    total_value = 0.0

    for ratio, value, weight in items:
        if capacity >= weight:
            capacity -= weight
            total_value += value
            print("Added full item: Value = ",value, "Weight = ",weight, "Capacity left = ",capacity)
        else:
            total_value += ratio * capacity
            print("Added fraction: Value = ",ratio * capacity, "Weight = ",capacity, "Capacity left = 0")
            break
    return total_value

values = [100, 60, 120] 
weights = [20, 10, 30]
capacity = 50

result = fractional_knapsack(capacity, weights, values)
print("Total value :", result)

<!----------------------------- 0-1 knapsack problem ------------------->

def knapsackRec(w, val, wt, n):
    if n == 0 or w == 0:
        return 0
    pick = 0
    if wt[n - 1] <= w:
        pick = val[n - 1] +knapsackRec(w - wt[n - 1], val, wt, n-1)
    notPick = knapsackRec(w, val,wt, n-1)
    if pick > notPick:
        print("Item ",n, " (Val:",val[n-1], ", Wt:",wt[n-1],"): Picked")
    else:
        print("Item ",n, " (Val:",val[n-1], ", Wt:",wt[n-1],"): Skipped")
    return max(pick, notPick)

def knapsack(w, val, wt):
    n = len(val)
    return knapsackRec(w, val, wt, n)

val = [1, 7, 11]
wt = [1, 2, 3]
w = 5
print("Steps:")
result = knapsack(w, val, wt)
print("Maximum Value:", result)

from collections import deque

def bfs(graph, start):
    visited = set()
    queue = deque([start])

    while queue:
        node = queue.popleft()
        if node not in visited:
            print(node, end=" ")
            visited.add(node)
	    #print("Visited: ",node," | Queue left: ",list(queue))

        for neighbor in graph[node]:
            if neighbor not in visited:
                queue.append(neighbor)
        
graph = {
 'A': ['B', 'C'],
 'B': ['A', 'D', 'E'],
 'C': ['A'],
 'D': ['B'],
 'E': ['B']
}
print("\nBFS Traversal:")
bfs(graph, 'A')

def dfs(graph, start, visited = None):
    if visited is None:
        visited = set()

    print(start, end = " ")
    visited.add(start)
    #print("Visited: ",start," | Visited set: ",visited)

    for neighbor in graph[start]:
        if neighbor not in visited:
            dfs(graph, neighbor, visited)

graph = {
    'A' : ['B','C'],
    'B' : ['A','D','E'],
    'C' : ['A'],
    'D' : ['B'],
    'E' : ['B']
}

print("DFS Traversal : ")
dfs(graph, 'A')


<!----------------------------- Matrix Chain Multiplication ------------------->


mc = [[-1 for n in range(50)] for m in range(50)]

def DynamicProgramming(c, i, j):
    if i == j:
        return 0
    if mc[i][j] != -1:
        return mc[i][j]
    mc[i][j] = 999999
    for k in range(i, j):
        mc[i][j] = min(mc[i][j], DynamicProgramming(c, i, k) + DynamicProgramming(c, k + 1, j) + c[i - 1] * c[k] * c[j])
        print("Calculated cost for matrices [",i," to ",j,"]:",mc[i][j])
    return mc[i][j]

def Matrix(c, n):
    i = 1
    j = n - 1
    return DynamicProgramming(c, i, j)

arr = [5, 10, 15, 20, 25]
n = len(arr)
result = Matrix(arr, n)
print("-" * 40)
print("Minimum number of multiplications is:", result)
